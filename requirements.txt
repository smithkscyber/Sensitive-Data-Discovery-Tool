# Versions are pinned so the synthetic corpus in data/raw/ stays stable across
# machines and over time. The answer key records character offsets, which are
# only valid for the exact text Faker produced -- an unpinned upgrade would
# silently invalidate every offset. Upgrade deliberately, then regenerate the
# corpus and the key together.
#
# .txt and .csv fixtures are byte-identical run to run. The .docx and .pdf
# fixtures carry fixed metadata timestamps, but their compressed bytes may
# still vary with library version; what is guaranteed stable is the text their
# parsers extract, which is what the offsets index.
#
# The same reasoning applies to the NLP stack: Presidio's accuracy comes from a
# spaCy model, and a model upgrade changes which entities are found and how
# their spans are bounded. Every precision/recall figure in the README is tied
# to these exact versions.
#
faker==40.39.0
pandas==3.0.5
presidio-analyzer==2.2.364
presidio-anonymizer==2.2.364
spacy==3.8.16
pdfplumber==0.11.10
python-docx==1.2.0

# Writes the .pdf fixtures in scripts/generate_test_data.py. Not needed to run
# the scanner -- only to regenerate the corpus.
fpdf2==2.8.8

# UI. altair comes with streamlit but is pinned because app.py imports it
# directly for the findings chart -- st.bar_chart orders categories
# alphabetically and cannot be told to sort by value.
streamlit==1.64.0
altair==6.3.0

# OCR, for documents that carry no text layer -- a scanned PDF is a picture of
# a page, and without this it extracts to nothing and reports as clean.
# pytesseract is a thin wrapper: Tesseract itself is a system binary and must
# be installed separately (see README). Everything degrades to "no text
# recovered" when it is absent rather than failing the scan.
pytesseract==0.3.13
pillow==12.3.0

# Spreadsheet parsing. .eml, .html and .json need nothing -- stdlib covers them.

# Property-based testing. tests/test_properties.py states invariants -- offsets
# address the text, the merge never emits overlapping spans, a redaction never
# leaks -- and hypothesis goes looking for inputs that break them, shrinking any
# counterexample to the smallest one that still fails. It found two: a
# one-character mailbox that redacted to itself, and a short value that "keep
# the last four" left untouched.
hypothesis==6.168.0

# Mutation testing. Scores the test suite rather than the code by breaking the
# detectors on purpose and checking the tests notice. Configured in setup.cfg;
# not needed to run or test the tool, only to audit how much the tests are
# really checking -- which is why it is the one package deliberately left out
# of requirements.lock.txt. It pulls in libcst, textual and playwright, a far
# larger tree than the thing it measures. Install it only when auditing.
mutmut==2.5.1
